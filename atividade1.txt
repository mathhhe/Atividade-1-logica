import java.util.Scanner;
public class atividade {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);
        int arrayresultado[] = new int[6];
        int quantidadeVitorias = 0;
        

        System.out.println("Digite o resulta dos jogos desse aluno(V = Vitória, P = Derrota): ");

        for (int i = 0; i < 6; i++) {
            String resultado = ler.nextLine();
            if (resultado.equalsIgnoreCase("V")) {
                arrayresultado[i] = 1;
            } else if (resultado.equalsIgnoreCase("P")) {
                arrayresultado[i] = 0;
            } else {
                System.out.println("Resultado inválido. Digite novamente.");
                i--;
            }
            if (arrayresultado[i] == 1) {
                quantidadeVitorias++;
            }

        }
        if(quantidadeVitorias >= 5 && quantidadeVitorias <= 6){
            System.out.println("grupo 1!");
        } else if(quantidadeVitorias >= 3 && quantidadeVitorias <= 4){
            System.out.println("grupo 2!");
        } else if(quantidadeVitorias >= 1 && quantidadeVitorias <= 2){
            System.out.println("grupo 1!");
        } else{
            System.out.println("grupo -1!");
        }
    }
       
}

import java.util.Scanner;
public class atividade {
    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);
        System.out.println("Digite um numero inteiro (S deve ser menor que A): ");
        int numeroS = ler.nextInt();
        System.out.println("Digite outro numero inteiro(A deve ser menor que B): ");
        int numeroA = ler.nextInt();
        System.out.println("Digite outro numero inteiro(B deve ser maior que A): ");
        int numeroB = ler.nextInt();

        int quantidade = 0;

        for (int i = numeroA; i <= numeroB; i++) {
           int primeirodigito = i / 10;
           int segundodigito = i % 10;
           if (primeirodigito + segundodigito == numeroS) {
               quantidade++;
           }
        }

        System.out.println("Quantidade de numeros que satisfazem a condicao: " + quantidade);


    }
       
}
